    #!/usr/bin/env bash
    # usage: rename-shots.sh [--go] DIR
    go=0; [ "$1" = "--go" ] && { go=1; shift; }
    i=1
    for f in "$1"/Screenshot*.png; do
      new="$1/shot-$(date -r "$f" +%Y%m%d)-$(printf '%02d' "$i").png"
      echo "$f -> $new"
      [ "$go" = 1 ] && mv -n "$f" "$new"
      i=$((i+1))
    done
